<script>
(function(){
  var root=document.querySelector('[data-cc-sheet]');
  if(!root||root.__sh){return}
  root.__sh=1;
  var panel=root.querySelector('.cc-sh__panel');
  var inner=root.querySelector('.cc-sh__in');
  var host=null,timer=null;
  var saleSel='.ux-hero .ux-btn,.ux-nav__item--sale > a,.bf-cb__link.is-hot,.cct-promo__text,.cct-promo__cta,.cc-mnav-promo,[data-sheet-open]';
  function top(){
    var b=0;
    document.querySelectorAll('.shopify-section-group-header-group').forEach(function(el){
      var r=el.getBoundingClientRect();
      if(r.height>0&&r.bottom>b){b=r.bottom}
    });
    return Math.max(0,Math.round(b));
  }
  function open(trigger){
    if(timer){clearTimeout(timer);timer=null}
    var h=trigger?trigger.closest('.ux-hero,[data-sheet-host]'):null;
    if(h&&window.innerWidth>=990){
      host=h;
      host.appendChild(root);
      root.classList.add('cc-sh--split');
    }else{
      host=null;
      document.body.appendChild(root);
      root.classList.remove('cc-sh--split');
      root.style.setProperty('--sh-top',(top()+8)+'px');
      document.documentElement.classList.add('cc-sh-lock');
    }
    root.hidden=false;
    if(host&&host.hasAttribute('data-sheet-host')){
      host.style.setProperty('--sh-h',Math.max(host.offsetHeight,inner.scrollHeight+8)+'px');
    }
    requestAnimationFrame(function(){requestAnimationFrame(function(){
      root.classList.add('is-open');
      if(host){host.classList.add('is-split-open')}
      try{panel.focus({preventScroll:true})}catch(e){}
    })});
  }
  function close(){
    if(root.hidden){return}
    root.classList.remove('is-open');
    var h=host;
    if(h){h.classList.remove('is-split-open')}
    document.documentElement.classList.remove('cc-sh-lock');
    timer=setTimeout(function(){
      root.hidden=true;
      if(h){document.body.appendChild(root);root.classList.remove('cc-sh--split')}
      host=null;
    },720);
  }
  document.addEventListener('click',function(e){
    var t=e.target;
    if(!t.closest){return}
    if(t.closest('[data-sh-close]')){e.preventDefault();e.stopPropagation();close();return}
    if(t.closest('.cc-sh__panel')){return}
    var a=t.closest(saleSel);
    if(a){
      if(e.metaKey||e.ctrlKey){return}
      e.preventDefault();e.stopPropagation();
      if(root.hidden){open(a)}else{close()}
      return;
    }
    if(!root.hidden){close()}
  },true);
  document.addEventListener('keydown',function(e){if(e.key==='Escape'){close()}});
})();
</script>
