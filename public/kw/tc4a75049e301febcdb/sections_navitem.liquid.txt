{%- comment -%}
  Desktop nav item with optional mega panel. Params: link, section, index
  Grouped mode (30-10): when menu children have their own links (e.g. Clothing / Shoes),
  every child becomes a titled column with its collections + counts.
  11-11-2026: grouped mode shows every collection as a photo card (first product image), same as the mobile menu.
  21-11-2026: photo cards also for a group without sub-links, and for simple (2-level) menus.
{%- endcomment -%}
{%- if link.links != blank -%}
  {%- liquid
    assign feature = nil
    for block in section.blocks
      if block.type == 'mega_feature'
        assign bt = block.settings.menu_title | strip | downcase
        assign lt = link.title | strip | downcase
        if bt == lt
          assign feature = block
          break
        endif
      endif
    endfor
    assign grouped = false
    for child in link.links
      if child.links != blank
        assign grouped = true
        break
      endif
    endfor
    assign shop_all = section.settings.shop_all_label | default: 'Shop all'
    assign styles_word = section.settings.styles_label | default: 'styles'
  -%}
  <li class="ux-nav__item ux-nav__item--mega" data-mega-item>
    <a class="ux-nav__link{% if link.child_active or link.active %} is-active{% endif %}" href="{{ link.url }}" aria-haspopup="true" aria-expanded="false" aria-controls="UxMega-{{ index }}">
      <span>{{ link.title }}</span>{% render 'ux-icon', icon: 'chevron-down', class: 'ux-nav__caret' %}
    </a>
    <div class="ux-mega" id="UxMega-{{ index }}">
      <div class="ux-page ux-mega__inner">
        <div class="ux-mega__col">
          <a class="ux-mega__all" href="{{ link.url }}">
            <span class="ux-mega__all-label">{{ shop_all | escape }} {{ link.title }}</span>
            {%- if section.settings.show_counts and link.type == 'collection_link' and link.object.products_count > 0 -%}
              <span class="ux-mega__all-count">{{ link.object.products_count }} {{ styles_word | escape }}</span>
            {%- endif -%}
            {% render 'ux-icon', icon: 'arrow', class: 'ux-mega__all-arrow' %}
          </a>
          <div class="ux-mega__groups cc-mega-groups">
            {%- if grouped -%}
              {%- comment -%} 25-11: one photo per group (block "Mega menu group photo"), sub-collections as text links under it {%- endcomment -%}
              {%- liquid
                assign lt = link.title | strip | downcase
                assign hot_word = section.settings.hot_word | strip | downcase
              -%}
              {%- for child in link.links -%}
                {%- liquid
                  assign gimg = nil
                  assign ct = child.title | strip | downcase
                  for gb in section.blocks
                    if gb.type == 'group_photo' and gb.settings.image != blank
                      assign bm = gb.settings.menu_title | strip | downcase
                      assign bg = gb.settings.group_title | strip | downcase
                      if bm == lt and bg == ct
                        assign gimg = gb.settings.image
                        break
                      endif
                    endif
                  endfor
                  if gimg == nil and child.type == 'collection_link'
                    assign gimg = child.object.image | default: child.object.products[forloop.index0].featured_image | default: child.object.products.first.featured_image
                  endif
                -%}
                <div class="ux-mega__group cc-mg">
                  <a class="cc-mg__head" href="{{ child.url }}">
                    <span class="cc-mg__img">{%- if gimg -%}{{ gimg | image_url: width: 480 | image_tag: loading: 'lazy', alt: child.title, sizes: '180px' }}{%- endif -%}</span>
                    <span class="cc-mg__title">{{ child.title | escape }}</span>
                  </a>
                  {%- if child.links != blank -%}
                    <ul class="cc-mg__links" role="list">
                      {%- for gc in child.links -%}
                        {%- liquid
                          assign gt = gc.title | downcase
                          assign is_hot = false
                          if hot_word != blank and gt contains hot_word
                            assign is_hot = true
                          endif
                        -%}
                        <li><a href="{{ gc.url }}"{% if is_hot %} class="is-hot"{% endif %}><span class="cc-mg__name">{{ gc.title | escape }}</span>{%- if is_hot and section.settings.hot_tag != blank -%}<span class="cc-mg__tag">{{ section.settings.hot_tag | escape }}</span>{%- elsif section.settings.show_counts and gc.type == 'collection_link' and gc.object.products_count > 0 -%}<span class="cc-mg__count">{{ gc.object.products_count }}</span>{%- endif -%}</a></li>
                      {%- endfor -%}
                    </ul>
                  {%- endif -%}
                </div>
              {%- endfor -%}
            {%- else -%}
              {%- liquid
                assign cols = link.links.size
                if cols > 6
                  assign cols = 6
                endif
              -%}
              <div class="ux-mega__group cc-mega-group">
                <p class="ux-mega__group-title">{{ section.settings.shop_all_label | default: 'Shop all' | escape }}</p>
                <ul class="cc-mega-chips" role="list" style="--cc-cols: {{ cols }}">
                  {%- for child in link.links -%}
                    <li>{% render 'cc-mega-chip', item: child, section: section %}</li>
                  {%- endfor -%}
                </ul>
              </div>
            {%- endif -%}
          </div>
        </div>
        {%- if feature -%}
          <a class="ux-mega__feature" href="{{ feature.settings.link | default: link.url }}" {{ feature.shopify_attributes }}>
            {%- if feature.settings.image != blank -%}
              {{ feature.settings.image | image_url: width: 1200 | image_tag: class: 'ux-mega__img', widths: '600, 900, 1200', sizes: '560px', loading: 'lazy', alt: feature.settings.heading }}
            {%- else -%}
              {{ 'lifestyle-1' | placeholder_svg_tag: 'ux-mega__img ux-placeholder' }}
            {%- endif -%}
            <span class="ux-mega__scrim"></span>
            <span class="ux-mega__overlay">
              {%- if feature.settings.eyebrow != blank -%}<span class="ux-mega__eyebrow">{{ feature.settings.eyebrow | escape }}</span>{%- endif -%}
              {%- if feature.settings.heading != blank -%}<span class="ux-mega__headline">{{ feature.settings.heading | escape }}</span>{%- endif -%}
              {%- if feature.settings.button_label != blank -%}<span class="ux-btn ux-btn--white ux-mega__cta">{{ feature.settings.button_label | escape }}{% render 'ux-icon', icon: 'arrow', class: 'ux-btn__arrow' %}</span>{%- endif -%}
            </span>
          </a>
        {%- endif -%}
      </div>
    </div>
  </li>
{%- else -%}
  <li class="ux-nav__item">
    <a class="ux-nav__link{% if link.active %} is-active{% endif %}" href="{{ link.url }}"><span>{{ link.title }}</span></a>
  </li>
{%- endif -%}
