{%- comment -%}
  Shop collection page (30-10) for Women's / Men's and their categories.
  Reads the category structure from the navigation menu (top item = gender, children = groups,
  grandchildren = category collections), so the menu, the tiles and the sidebar always match.
{%- endcomment -%}
{%- liquid
  assign s = section.settings
  assign per_page = s.per_page
  assign sort_by = collection.sort_by | default: collection.default_sort_by
  assign nav = nil
  for top in s.menu.links
    if top.url == collection.url
      assign nav = top
      break
    endif
    for child in top.links
      if child.links == blank and child.url == collection.url
        assign nav = top
      endif
      for gc in child.links
        if gc.url == collection.url
          assign nav = top
        endif
      endfor
    endfor
    if nav
      break
    endif
  endfor
  assign is_landing = false
  if nav and nav.url == collection.url
    assign is_landing = true
  endif
  assign banner = nil
  if nav
    for block in section.blocks
      if block.type == 'banner' and block.settings.collection != blank and block.settings.collection.url == nav.url
        assign banner = block
        break
      endif
    endfor
  endif
  assign active_count = 0
  for filter in collection.filters
    assign active_count = active_count | plus: filter.active_values.size
    if filter.type == 'price_range'
      if filter.min_value.value != nil or filter.max_value.value != nil
        assign active_count = active_count | plus: 1
      endif
    endif
  endfor
-%}

<script>
  (function () {
    var root = document.querySelector('#shopify-section-{{ section.id }} [data-cats]');
    if (!root) return;
    var track = root.querySelector('[data-cats-track]');
    var prev = root.querySelector('[data-cats-prev]');
    var next = root.querySelector('[data-cats-next]');
    var sync = function () {
      var max = track.scrollWidth - track.clientWidth - 2;
      root.classList.toggle('has-overflow', max > 0);
      prev.disabled = track.scrollLeft <= 2;
      next.disabled = track.scrollLeft >= max;
    };
    prev.addEventListener('click', function () { track.scrollBy({ left: -track.clientWidth * 0.8, behavior: 'smooth' }); });
    next.addEventListener('click', function () { track.scrollBy({ left: track.clientWidth * 0.8, behavior: 'smooth' }); });
    track.addEventListener('scroll', sync, { passive: true });
    window.addEventListener('resize', sync);
    var active = track.querySelector('.is-active');
    if (active && active.parentElement.offsetLeft > track.clientWidth * 0.6) {
      track.scrollLeft = active.parentElement.offsetLeft - 40;
    }
    sync();
  })();
</script>
