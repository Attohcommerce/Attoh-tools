<script>
  (function(){
    document.querySelectorAll('[data-bf-cb]').forEach(function(bar){
      if (bar.dataset.ready) return; bar.dataset.ready = '1';
      var nav = bar.querySelector('.bf-cb__nav');
      function upd(){
        var over = nav.scrollWidth > nav.clientWidth + 4;
        bar.classList.toggle('is-scroll', over);
        bar.classList.toggle('at-start', nav.scrollLeft < 4);
        bar.classList.toggle('at-end', nav.scrollLeft + nav.clientWidth > nav.scrollWidth - 4);
      }
      bar.querySelectorAll('.bf-cb__arrow').forEach(function(b){
        b.addEventListener('click', function(){ nav.scrollBy({ left: Number(b.dataset.dir) * nav.clientWidth * 0.7, behavior: 'smooth' }); });
      });
      nav.addEventListener('scroll', upd, { passive: true });
      window.addEventListener('resize', upd);
      upd(); setTimeout(upd, 300);
    });
  })();
</script>
