<script>
(function(){
  var d=document.getElementById('CcmDrawer'); if(!d) return;
  if(d.parentNode!==document.body) document.body.appendChild(d);
  function open(){d.classList.add('is-open');d.setAttribute('aria-hidden','false');document.documentElement.classList.add('ccm-lock');}
  function close(){d.classList.remove('is-open');d.setAttribute('aria-hidden','true');document.documentElement.classList.remove('ccm-lock');}
  var sel='[data-drawer-open],.ux-header__burger,[aria-controls="UxNavDrawer"],[data-ccm-open]';
  document.addEventListener('click',function(e){
    var t=e.target.closest(sel); if(!t) return;
    e.preventDefault(); e.stopImmediatePropagation(); open();
  },true);
  d.addEventListener('click',function(e){ if(e.target.closest('[data-ccm-close]')) close(); });
  document.addEventListener('keydown',function(e){ if(e.key==='Escape') close(); });
})();
</script>
